// FINAL RUST VERSION — perfectly scaled for 800×480 display
// Layout:
//   40 px  – top bar
//   240 px – waterfall
//   100 px – spectrum
//   75 px  – trend graphs
//   Total = 455 (fits in 480 safely with margins)

use rustfft::{FftPlanner, num_complex::Complex};
use cpal::traits::{DeviceTrait, HostTrait, StreamTrait};
use sdl2::{pixels::Color, rect::Rect};
use std::sync::{Arc, Mutex};
use std::collections::VecDeque;
use std::path::Path;

// ==== CONSTANTS ====
const FFT_SIZE: usize = 2048;
const TOP_H: u32 = 40;
const WF_H: u32 = 240;
const SPEC_H: u32 = 100;
const TREND_H: u32 = 75;
const PEAK_MAX_HZ: f32 = 800.0;

// ==== FONT FINDER ====
fn find_font_path() -> Option<String> {
    let fonts = [
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/freefont/FreeSans.ttf",
        "/usr/share/fonts/truetype/liberation/LiberationSans-Regular.ttf",
        "/usr/share/fonts/truetype/ubuntu/Ubuntu-R.ttf",
        "./DejaVuSans.ttf"
    ];
    for p in fonts {
        if Path::new(p).exists() {
            return Some(p.to_string());
        }
    }
    None
}

// ==== AUDIO STATS ====
struct AudioStats {
    fft: VecDeque<f32>,
    peak: VecDeque<f32>,
}
impl AudioStats {
    fn new() -> Self {
        Self {
            fft: VecDeque::with_capacity(300),
            peak: VecDeque::with_capacity(300),
        }
    }
    fn update(&mut self, t: f32, p: f32) {
        if self.fft.len() >= 300 {
            self.fft.pop_front();
            self.peak.pop_front();
        }
        self.fft.push_back(t);
        self.peak.push_back(p);
    }
}

// ==== DSP ====
fn hann(i: usize) -> f32 {
    0.5 * (1.0 - (2.0 * std::f32::consts::PI * i as f32 / FFT_SIZE as f32).cos())
}
fn to_db(x: f32) -> f32 {
    20.0 * (x.max(1e-10)).log10()
}

// ==== MAIN ====
fn main() {
    // SDL2
    let sdl = sdl2::init().unwrap();
    let video = sdl.video().unwrap();
    let dm = video.current_display_mode(0).unwrap();
    let width = dm.w as u32;
    let height = dm.h as u32;

    let window = video
        .window("Rust FFT Visualizer", width, height)
        .borderless()
        .build()
        .unwrap();

    let mut canvas = window.into_canvas().accelerated().present_vsync().build().unwrap();
    let tc = canvas.texture_creator();

    // TTF
    let ttf = sdl2::ttf::init().unwrap();
    let font_path = find_font_path().expect("Font not found");
    let font = ttf.load_font(&font_path, 18).unwrap();
    let small = ttf.load_font(&font_path, 14).unwrap();

    // AUDIO
    let host = cpal::default_host();
    let device = host.default_input_device().expect("No mic");
    let config = device.default_input_config().unwrap();
    let sr = config.sample_rate().0 as f32;
    let config = config.into();

    let buf = Arc::new(Mutex::new(vec![0.0f32; FFT_SIZE]));
    let buf2 = buf.clone();

    let stream = device.build_input_stream(
        &config,
        move |data: &[f32], _| {
            let mut b = buf2.lock().unwrap();
            let n = data.len().min(FFT_SIZE);
            b[..n].copy_from_slice(&data[..n]);
        },
        move |err| eprintln!("Audio error: {}", err),
        None
    ).unwrap();
    stream.play().unwrap();

    // FFT
    let mut planner = FftPlanner::new();
    let fft = planner.plan_fft_forward(FFT_SIZE);
    let mut spec = vec![Complex{re:0.0,im:0.0}; FFT_SIZE];

    // VISUAL BUFFERS
    let mut wf = vec![vec![0.0; FFT_SIZE/2]; WF_H as usize];
    let mut prev = vec![0.0; FFT_SIZE/2];
    let mut stats = AudioStats::new();
    let mut pump = sdl.event_pump().unwrap();

    let x_scale = width as f32 / (FFT_SIZE as f32 / 2.0);

    // ==== MAIN LOOP ====
    'main: loop {
        // Events
        for e in pump.poll_iter() {
            use sdl2::event::Event::*;
            match e {
                Quit{..} => break 'main,
                KeyDown{keycode:Some(k),..} if k == sdl2::keyboard::Keycode::Escape => break 'main,
                _ => {}
            }
        }

        // Acquire audio
        let b = buf.lock().unwrap();
        let mut win = vec![0.0f32; FFT_SIZE];
        for i in 0..FFT_SIZE {
            win[i] = b[i] * hann(i);
        }
        drop(b);

        // Fill FFT buffer
        for i in 0..FFT_SIZE {
            spec[i].re = win[i];
            spec[i].im = 0.0;
        }

        // Run FFT
        let t0 = std::time::Instant::now();
        fft.process(&mut spec);
        let fft_ms = t0.elapsed().as_micros() as f32 / 1000.0;

        // Magnitudes
        let mut mags = Vec::with_capacity(FFT_SIZE/2);
        for c in spec.iter().take(FFT_SIZE/2) {
            let db = to_db(c.norm());
            mags.push(((db + 80.0) / 80.0).clamp(0.0, 1.0));
        }

        // Smooth
        for i in 0..mags.len() { mags[i] = 0.3*mags[i] + 0.7*prev[i]; }
        prev.copy_from_slice(&mags);

        // Peak frequency
        let (pi, _) = mags.iter().enumerate().max_by(|(_,a),(_,b)| a.partial_cmp(b).unwrap()).unwrap();
        let peak_hz = pi as f32 * sr / FFT_SIZE as f32;
        stats.update(fft_ms, peak_hz);

        // Waterfall scroll
        wf.remove(0);
        wf.push(mags.clone());

        // ========================================================
        // DRAW BEGIN
        // ========================================================
        canvas.set_draw_color(Color::RGB(15,15,20));
        canvas.clear();

        // ==== TOP BAR ====
        let title = font.render("RUST SIGNAL MODEL — SAFE • FAST • REALTIME")
            .blended(Color::RGB(255,255,255)).unwrap();
        let ttex = tc.create_texture_from_surface(&title).unwrap();
        canvas.copy(&ttex, None, Rect::new(8,4,title.width(),title.height())).ok();

        let rust_line = small
            .render("Zero memory leaks · No null pointers · No buffer overruns · Safe concurrency")
            .blended(Color::RGB(180,200,255)).unwrap();
        let rtex = tc.create_texture_from_surface(&rust_line).unwrap();
        canvas.copy(&rtex, None, Rect::new(8,22,rust_line.width(),rust_line.height())).ok();

        // ==== WATERFALL ====
        for row in 0..WF_H as usize {
            let y = TOP_H as i32 + row as i32;
            for (i,v) in wf[row].iter().enumerate() {
                let col = if *v < 0.33 {
                    let t = *v / 0.33;
                    Color::RGB(0, (t*255.0) as u8, 255)
                } else if *v < 0.66 {
                    let t = (*v - 0.33) / 0.33;
                    Color::RGB((t*255.0) as u8, 255, (255.0*(1.0-t)) as u8)
                } else {
                    let t = (*v - 0.66) / 0.34;
                    Color::RGB(255, (255.0*(1.0-t)) as u8, 0)
                };
                canvas.set_draw_color(col);
                let x = (i as f32 * x_scale) as i32;
                canvas.fill_rect(Rect::new(x, y, x_scale.ceil() as u32, 1)).ok();
            }
        }

        // ==== SPECTRUM ====
        let sy = TOP_H as i32 + WF_H as i32;
        canvas.set_draw_color(Color::RGB(20,20,25));
        canvas.fill_rect(Rect::new(0, sy, width, SPEC_H)).ok();

        for (i,v) in mags.iter().enumerate() {
            let h = (v * SPEC_H as f32) as i32;
            let x = (i as f32 * x_scale) as i32;
            let col = if i == pi { Color::RGB(255,50,50) }
            else { Color::RGB(0,200,255) };
            canvas.set_draw_color(col);
            canvas.fill_rect(Rect::new(x, sy + SPEC_H as i32 - h, x_scale.ceil() as u32, h as u32)).ok();
        }

        // ==== TRENDS ====
        let ty = sy + SPEC_H as i32 - TREND_H as i32 + 2;
        let half = width / 2;
        let gh = TREND_H - 18;

        // LEFT: FFT TIME
        let left = Rect::new(0, ty, half, TREND_H);
        canvas.set_draw_color(Color::RGB(20,20,25)); canvas.fill_rect(left).ok();
        canvas.set_draw_color(Color::RGB(80,80,80)); canvas.draw_rect(left).ok();

        let label = small.render("FFT Time (ms)").blended(Color::RGB(0,255,0)).unwrap();
        let ltex = tc.create_texture_from_surface(&label).unwrap();
        canvas.copy(&ltex,None,Rect::new(4,ty+2,label.width(),label.height())).ok();

        if stats.fft.len() > 1 {
            let max = stats.fft.iter().cloned().fold(0.0,f32::max).max(0.1);
            canvas.set_draw_color(Color::RGB(0,255,0));
            let n = stats.fft.len();
            for i in 0..n-1 {
                let x1 = 2 + (i * (half as usize - 4) / 300) as i32;
                let x2 = 2 + ((i+1)*(half as usize - 4)/300) as i32;
                let y1 = ty + TREND_H as i32 - 2 - ((stats.fft[i]/max)*(gh as f32)) as i32;
                let y2 = ty + TREND_H as i32 - 2 - ((stats.fft[i+1]/max)*(gh as f32)) as i32;
                canvas.draw_line((x1,y1),(x2,y2)).ok();
            }
        }

        // RIGHT: PEAK FREQ (0–800 Hz, no overshoot)
        let right = Rect::new(half as i32, ty, half, TREND_H);
        canvas.set_draw_color(Color::RGB(20,20,25)); canvas.fill_rect(right).ok();
        canvas.set_draw_color(Color::RGB(80,80,80)); canvas.draw_rect(right).ok();

        let flabel = small.render("Peak Frequency (Hz)").blended(Color::RGB(255,220,0)).unwrap();
        let ftex = tc.create_texture_from_surface(&flabel).unwrap();
        canvas.copy(&ftex,None,Rect::new(half as i32+4,ty+2,flabel.width(),flabel.height())).ok();

        if stats.peak.len() > 1 {
            canvas.set_draw_color(Color::RGB(255,200,50));
            let n = stats.peak.len();
            for i in 0..n-1 {
                let f1 = stats.peak[i].clamp(0.0, PEAK_MAX_HZ);
                let f2 = stats.peak[i+1].clamp(0.0, PEAK_MAX_HZ);

                let x1 = half as i32 + 2 + (i*(half as usize - 4)/300) as i32;
                let x2 = half as i32 + 2 + ((i+1)*(half as usize - 4)/300) as i32;

                let y1 = ty + TREND_H as i32 - 2 - ((f1 / PEAK_MAX_HZ) * gh as f32) as i32;
                let y2 = ty + TREND_H as i32 - 2 - ((f2 / PEAK_MAX_HZ) * gh as f32) as i32;

                canvas.draw_line((x1,y1),(x2,y2)).ok();
            }
        }

        canvas.present();
    }
}
